```python
import sys
from playwright.sync_api import sync_playwright

with open(sys.argv[1], encoding='utf-8') as f:
    prompt = f.read().strip()

with sync_playwright() as p:
    browser = p.chromium.connect_over_cdp("http://127.0.0.1:9222")
    target = None
    for ctx in browser.contexts:
        for pg in ctx.pages:
            if 'iwencai' in pg.url and '/chat' in pg.url:
                target = pg
                break
        if target:
            break

    editor = target.query_selector('[contenteditable], .ql-editor')
    target.evaluate('''(text) => {
        const el = document.querySelector('[contenteditable]');
        el.innerHTML = '';
        el.textContent = text;
    }''', prompt)

    target.query_selector('.send-button').click()
    target.wait_for_timeout(35000)

    response = target.evaluate('() => document.body.innerText')
    with open(sys.argv[1].replace('_prompt.md', '_response.md'), 'w', encoding='utf-8') as f:
        f.write(response)
```

### 4.4.1 `scripts/step3_5m_search.py`（91 行，全文）—— 并行知几 search

```python
#!/usr/bin/env python3
"""Step3-5M: 并行知几 search (138节点/1102唯一指标)。

输入: /tmp/decision_indicators_5m.json  {CODE:{node:[指标名]}}
输出: analysis/iwencai/step3_search_results_5m.json
      {CODE:{指标名:{nodes,hits:[{id,name,source,unit,path}],count}}}

分批提交(consumer 立即消费, 避免 1102 任务全堆积); 断点续跑。
并发 4, 批间隔 0.3s。
"""
import json, os, re, subprocess, sys, time
from concurrent.futures import ThreadPoolExecutor

ROOT = "/home/ubuntu/framework-tree/analysis/iwencai"
SRC = "/tmp/decision_indicators_5m.json"
OUT = os.path.join(ROOT, "step3_search_results_5m.json")
ZJ = os.path.expanduser("~/.hermes/scripts/zhiji_api.py")
CODES = ["ZN", "NI", "SN", "SI", "LI"]
BATCH = 40
BATCH_GAP = 0.3

def clean(name):
    n = re.sub(r"【[^】]*】", "", name).strip()
    return re.sub(r"^[*-]\s*", "", n).strip()

def search_one(code, q):
    try:
        r = subprocess.run([sys.executable, ZJ, "search", q],
                           capture_output=True, text=True, timeout=60)
        j = json.loads(r.stdout.strip())
        hits = j.get("results", [])[:3]
        return {"nodes": q_nodes[code][q],
                "hits": [{"id": h.get("id"), "name": h.get("name"),
                          "source": h.get("source"), "unit": h.get("unit"),
                          "path": (h.get("path") or "")[:60]} for h in hits],
                "count": j.get("count")}
    except Exception as e:
        return {"nodes": q_nodes[code][q], "hits": [], "error": str(e)[:200]}

q_nodes = {}

def main():
    data = json.load(open(SRC, encoding="utf-8"))
    for code in CODES:
        uniq = {}
        for node, names in data[code].items():
            for n in names:
                c = clean(n)
                if c and len(c) >= 2:
                    uniq.setdefault(c, []).append(node)
        q_nodes[code] = uniq

    results = json.load(open(OUT, encoding="utf-8")) if os.path.exists(OUT) else {}
    for code in CODES:
        results.setdefault(code, {})

    todo = [(code, q) for code in CODES for q in q_nodes[code]
            if q not in results[code]]
    print("待搜: %d" % len(todo), flush=True)
    if not todo:
        return

    t0 = time.time()
    done = 0
    with ThreadPoolExecutor(max_workers=4) as ex:
        for start in range(0, len(todo), BATCH):
            batch = todo[start:start + BATCH]
            futs = {ex.submit(search_one, c, q): (c, q) for c, q in batch}
            for f in futs:
                c, q = futs[f]
                results[c][q] = f.result()
                done += 1
            json.dump(results, open(OUT, "w"), ensure_ascii=False, indent=1)
            if done % 200 == 0 or done == len(todo):
                el = time.time() - t0
                print("%d/%d  %.0fs  avg %.2fs/项" % (done, len(todo), el, el / done),
                      flush=True)
            time.sleep(BATCH_GAP)

    print("\n=== 汇总 ===")
    tot_hit = 0
    for code in CODES:
        d = results[code]
        zero = sum(1 for v in d.values() if not v.get("hits"))
        tot_hit += len(d) - zero
        print("%s: %d 指标, 零命中 %d" % (code, len(d), zero), flush=True)
    print("有命中合计: %d" % tot_hit)
    print("完成 ->", OUT)

if __name__ == "__main__":
    main()
```

```python
    "NI": ["锌", "锡", "硅", "锂", "铝", "铜", "铅"],

    "LI": ["锌", "镍", "锡", "铝", "铜", "铅"],

def other_var_bad(q, hname, code):
    """命中名含其他品种词 → 串台淘汰 (如 锡现货升贴水 命中 电解铝升贴水)。"""
    for w in OTHER_VAR.get(code, []):
        if w in hname:
            # 豁免: 硅→硅铁/硅锰 属硅系; 锂→氢氧化锂
            if code == "SI" and w in ("铝",) and "硅" in hname:
                continue
            return True
    return False


def score_hit(q, h, code):
    hname = h.get("name", "") or ""
    wid = h.get("id", "") or ""
    s = 0
    vws = VAR_WORDS.get(code, [])
    # 品种词
    if any(w in hname for w in vws):
        s += 2
    elif any(w.lower() in wid.lower() for w in vws if len(w) >= 2):
        s += 1
    else:
        return -1
    # 字段词
    fields = [f for f in FIELD_WORDS if f in q]
    s += sum(2 for f in fields if f in hname)
    # 交易所前缀
    for pre in EXCHANGES:
        if pre in q and pre not in hname and pre not in wid:
            s -= 3
    # 口径陷阱 / 地理淘汰 / 跨品种串台
    if minus_bad(q, hname, code):
        return -2
    if geo_bad(q, hname):
        return -2
    if other_var_bad(q, hname, code):
        return -2
    # 矿端/冶炼端/下游 串台: 软惩罚 (-3, 阈值5下通常直接出局)
    for stage, allowed in STAGE.items():
        if stage in q and not any(a in hname for a in allowed):
            s -= 3
    # 官方源
    if h.get("source") in ("smm", "mysteel"):
        s += 1
    return s


def judge(q, v, code):
    hits = v.get("hits", [])
    if in_design(q):
        return {"matched": False, "chosen": None,
                "note": "设计说明/口径注释, 非可检索指标", "hits": hits}
    if not hits:
        return {"matched": False, "chosen": None,
                "note": "知几 search 无任何命中", "hits": hits}
    scored = []
    for h in hits:
        sc = score_hit(q, h, code)
        if sc >= 0:
            scored.append((sc, h))
    scored.sort(key=lambda x: -x[0])
    if scored and scored[0][0] >= 5:
        sc, best = scored[0]
        return {"matched": True, "chosen": best,
                "note": "命中%d条, 最佳得分%d: %s" % (len(scored), sc,
                                                     str(best.get("name", ""))[:50]),
                "hits": [h for _, h in scored]}
    top = scored[0] if scored else None
    return {"matched": False, "chosen": None,
            "note": "存在通过品种词但字段/口径弱 (最佳得分%d, 阈5): %s" % (
                top[0] if top else 0,
                str(top[1].get("name", ""))[:50] if top else "无"),
            "hits": hits}


def main():
    data = json.load(open(SRC, encoding="utf-8"))
    os.makedirs(os.path.dirname(OUT), exist_ok=True)
    out = {}
    for code in CODES:
        out[code] = {q: judge(q, v, code) for q, v in data.get(code, {}).items()}
        m = sum(1 for x in out[code].values() if x["matched"])
        print("%s: matched %d/%d (%.0f%%)" % (code, m, len(out[code]),
                                               100.0 * m / max(1, len(out[code]))))
    json.dump(out, open(OUT, "w"), ensure_ascii=False, indent=1)
    print("->", OUT)
    tot = sum(len(out[c]) for c in CODES)
    tm = sum(1 for c in CODES for x in out[c].values() if x["matched"])
    print("合计 matched %d/%d" % (tm, tot))


if __name__ == "__main__":
    main()
```

### 4.4.6 `_verify_norm.py` —— 节点码三方一致检查（门禁补盲）

```python
# -*- coding: utf-8 -*-
"""复核 divergence 规范化质量。用法: python _verify_norm.py NI ZN [SI SN LI]"""
import glob, re, sys

PROMPT_START = "角色：你是有色金属产业研究的"
RECO_TAIL = re.compile(r"以下内容[:：]\s*$")
STATUS_PAT = re.compile(r"^\s*(问财\s*模型|快速推理中)")
SUMMARY_PAT = re.compile(r"本维度（[^）]+）合计\s*\d+\s*个直接相关指标")
RECO_ITEM = re.compile(r"自选股中|重仓股战法|大盘技术分析|高roe|高毛利率战法|估值情况")
# 内容一致性：文件名/header/正文首行的节点码三者必须一致。
# 只查格式的 check_divergence 抓不到「答了别的节点」——
# LI 批次 4.5~7.3 共 11 个文件全是 4.4 的答案，check_divergence --strict 照样 exit=0。
NODE_HDR = re.compile(r"^#\s*([A-Z]{2})·.*?·([\d][\d.]*)\s")
NODE_BODY = re.compile(r"^(\d+\.\d+(?:\.\d+)?)\s")

allok = True
for v in sys.argv[1:]:
    fs = sorted(glob.glob(f"analysis/iwencai/{v}/divergence_*.md"))
    print(f"[{v}] 文件数: {len(fs)}")
    if not fs:
        print("  (无文件)"); continue
    bad, pipes, no_summary = [], [], []
    for f in fs:
        t = open(f, encoding="utf-8").read()
        ls = t.splitlines()
        name = f.split("/")[-1]
        pipe = sum(1 for l in ls if l.startswith("|"))
        tab = sum(1 for l in ls if "\t" in l)
        pipes.append(pipe)
        issues = []
        if pipe < 4: issues.append("pipe<4")
        if tab > 0: issues.append("tab残留")
        if "排除项" not in t: issues.append("缺排除项")
        if t.count("**排除项**") > 1: issues.append(f"排除项重复x{t.count('排除项')}")
        if PROMPT_START in t: issues.append("prompt回显")
        if any(RECO_TAIL.search(l) for l in ls): issues.append("推荐引导语残留")
        if any(RECO_ITEM.search(l) for l in ls): issues.append("推荐条目残留")
        if "AI复盘" in t or "智能图说" in t: issues.append("chrome残留")
        if "模型答案生成完成" in t: issues.append("完成标记残留")
        if any(STATUS_PAT.match(l) for l in ls): issues.append("状态行残留")
        # 内容一致性：文件名 ↔ header ↔ 正文首行 节点码三方对齐
        fn_node = re.search(r"divergence_([\d][\d.]*)\.md$", name).group(1)
        mh = NODE_HDR.match(ls[0])
        if not mh:
            issues.append("header缺节点码")
        elif mh.group(2) != fn_node:
            issues.append(f"header节点码{mh.group(2)}≠文件名{fn_node}")
        body_first = ""
        for l in ls[4:]:                      # 跳过 header(#/#/空/---) 与空行
            if l.strip():
                body_first = l.strip(); break
        mb = NODE_BODY.match(body_first)
        if not mb:
            issues.append("正文首行无节点码")
        elif mb.group(1) != fn_node:
            issues.append(f"答错节点:{mb.group(1)}≠{fn_node}")
        if not SUMMARY_PAT.search(t): no_summary.append(name)
        if issues: bad.append((name, ",".join(issues)))
    print("  不合格:", bad if bad else "无")
    wrong = [n for n, s in bad if s.startswith("答错节点")]
    if wrong:
        print(f"  ⚠️ 答错节点 {len(wrong)} 个: {', '.join(wrong)}")
    print(f"  管道表行范围: {min(pipes)}-{max(pipes)}")
    print(f"  平均字符数: {sum(len(open(f, encoding='utf-8').read()) for f in fs) // len(fs)}")
    print(f"  无本维度汇总行(参考, AI 写法差异): {len(no_summary)}/{len(fs)}")
    if bad: allok = False

print("\n=== 总结:", "全部合格" if allok else "有问题需处理", "===")
```

### 4.4.7 `_renorm.py` —— 规范化合并（prompt 回显提取）

```python
# -*- coding: utf-8 -*-
"""用改进后的 normalize_reply 就地重抽已抓取的 divergence 文件（不重新查询同花顺）。

被污染的 ZN 文件里仍保留完整原文（上轮答案 + 本轮 prompt 回显 + 本轮答案 + 推荐区），
新逻辑锚定「最后一次发出的 prompt」即可把本轮答案干净抽出来。
用法: python _renorm.py ZN [NI]
"""
import glob, importlib.util, re, sys

spec = importlib.util.spec_from_file_location("drv", "scripts/iwencai_batch_driver.py")
drv = importlib.util.module_from_spec(spec)
spec.loader.exec_module(drv)

SEP = "\n---\n\n"
changed, same, failed = [], [], []

for v in sys.argv[1:]:
    for f in sorted(glob.glob(f"analysis/iwencai/{v}/divergence_*.md")):
        txt = open(f, encoding="utf-8").read()
        i = txt.find(SEP)
        if i < 0:
            failed.append((f, "无 header 分隔")); continue
        head, body = txt[:i], txt[i + len(SEP):]
        # 幂等：剥掉上一轮追加的排除项块，避免重复追加
        body = re.sub(r"\n---\n\n> \*\*排除项\*\*.*\Z", "\n", body, flags=re.S)
        new_body = drv.normalize_reply(body)
        new_txt = head + SEP + new_body
        if new_txt != txt:
            changed.append((f.split("/")[-1], len(txt), len(new_txt)))
            open(f, "w", encoding="utf-8").write(new_txt)
        else:
            same.append(f.split("/")[-1])

print(f"改写: {len(changed)}  未变: {len(same)}  失败: {len(failed)}")
for n, a, b in changed:
    print(f"  {n:26s} {a} -> {b} 字符")
```
