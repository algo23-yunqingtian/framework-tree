### 4.2 RD-002详细分析

**问题描述**: Panel 63显示LME铜价9,850美元/吨，DSHB铜价72,300元/吨。按汇率7.00转换后为68,950元/吨，偏差3,350元(4.6%)。实际汇率为7.25，正确转换后为71,412.5元/吨，偏差887.5元(1.24%)。

1. 前端汇率缓存使用旧值(7.00)，实际汇率已更新为7.25

1. 汇率API调用改为实时获取
2. 汇率缓存TTL从24h缩短至1h
3. 汇率变更时触发前端刷新

**修复效果**:

|------|--------|--------|
| 汇率偏差 | 3.57% | 0% |
| 价格偏差 | 4.6% | 1.24% |

### 6.2 校验规则实现

#### RR-01: 数值精度校验

```javascript
// 校验规则: 图表值与API值偏差≤0.001%
function validateNumericalPrecision(chartValue, apiValue) {
  if (apiValue === null || apiValue === undefined) return { pass: true, reason: 'NULL值跳过' };
  if (apiValue === 0) {
    return { pass: Math.abs(chartValue) <= 0.001, reason: `偏差: ${chartValue}` };
  }
  const deviation = Math.abs(chartValue - apiValue) / Math.abs(apiValue);
  return {
    pass: deviation <= 0.00001, // 0.001%
    reason: `偏差: ${(deviation * 100).toFixed(6)}%`
  };
}
```

#### RR-02: 小数位校验

```javascript
// 校验规则: 前端显示小数位与配置一致
function validateDecimals(displayValue, configDecimals) {
  const str = displayValue.toString();
  const decimalPart = str.split('.')[1] || '';
  return {
    pass: decimalPart.length <= configDecimals,
    reason: `配置: ${configDecimals}位, 实际: ${decimalPart.length}位`
  };
}
```

#### RR-03: 单位转换校验

```javascript
// 校验规则: 单位转换正确
function validateUnitConversion(value, expectedUnit, actualUnit, conversionFormula) {
  const converted = conversionFormula(value, expectedUnit);
  return {
    pass: actualUnit === expectedUnit && Math.abs(converted - value) < 0.001,
    reason: `期望: ${expectedUnit}, 实际: ${actualUnit}`
  };
}
```

#### RR-04: 时间戳对齐校验

```javascript
// 校验规则: 时区/时间戳偏差≤1s

}
```

#### RR-05: 数据聚合校验

```javascript
// 校验规则: 聚合值与原始值偏差≤0.01%
function validateAggregation(aggregatedValue, rawValues) {
  const expected = rawValues.reduce((a, b) => a + b, 0) / rawValues.length;
  const deviation = Math.abs(aggregatedValue - expected) / Math.abs(expected);
  return {
    pass: deviation <= 0.0001, // 0.01%
    reason: `偏差: ${(deviation * 100).toFixed(6)}%`
  };
}
```

```javascript
// 校验规则: NULL/NaN正确断点处理
function validateMissingData(dataPoint, index, dataPoints) {
  if (dataPoint === null || dataPoint === undefined || Number.isNaN(dataPoint)) {
    // 检查前后数据点是否存在
    const prev = index > 0 ? dataPoints[index - 1] : null;

    return {
      pass: true, // 断点是正确的

  return { pass: true, reason: '非缺失数据' };
}
```
