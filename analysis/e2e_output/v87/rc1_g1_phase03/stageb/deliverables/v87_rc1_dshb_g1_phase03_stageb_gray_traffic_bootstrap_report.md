### 4.2 路由规则配置

```yaml
# dshb-router-v87-rc1.yaml
router:
  name: dshb-router-v87-rc1
  version: 2.4.0-rc1
  mode: gradual_rollout
  total_traffic: 15000  # ev/s

  routes:
    gray:
      target: gray-dshb-prod:8080
      version: v87.0.0-rc1
      ratio: 50
      health_check:
        interval: 1s
        timeout: 5s
        failure_threshold: 3
    baseline:
      target: base-dshb-prod:8080
      version: v86.0.0-rc2-frozen
      ratio: 50
      health_check:
        interval: 1s
        timeout: 5s
        failure_threshold: 3

  gradual_schedule:
    - stage: 5
      duration: 30s
    - stage: 20
      duration: 30s
    - stage: 50
      duration: continuous
    - stage: 75  # Phase03 StageC
    - stage: 100 # Phase03 StageD

  circuit_breaker:
    red_threshold: 3_metrics_continuous_30s
    action: rollback_to_baseline
    auto_resume: false
```
