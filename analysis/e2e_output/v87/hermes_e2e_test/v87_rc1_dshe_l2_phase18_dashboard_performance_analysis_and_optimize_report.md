| 365天 (3天) | 525,600 | 1,752 | 300x | 98.0% | 520ms |

### 5.2 下采样算法

```python
# 时序下采样算法 (LTTB - Largest Triangle Three Buckets)
def lttb_downsample(data, target_points):
    """
    LTTB算法保持数据趋势特征, 避免简单采样丢失极值
    
    参数:
        data: 原始时序数据
        target_points: 目标数据点数
    
    返回:
        下采样后数据
    """
    if len(data) <= target_points:
        return data
    
    sample_size = int(len(data) / target_points)
    sampled = [data[0]]  # 始终包含第一个点
    
    for i in range(1, target_points):
        start = i * sample_size
        end = min(start + sample_size, len(data))
        
        # 找到当前区间中面积最大的点
        max_area = -1
        best_idx = start
        for idx in range(start, end):
            area = abs(data[idx-1].y * data[idx].x - data[idx-1].x * data[idx].y)
            if area > max_area:
                max_area = area
                best_idx = idx
        
        sampled.append(data[best_idx])
    
    sampled.append(data[-1])  # 始终包含最后一个点
    return sampled
```
