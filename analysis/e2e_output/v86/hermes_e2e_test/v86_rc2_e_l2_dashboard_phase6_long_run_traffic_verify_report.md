def state_machine_continuity_monitor():
    """监控状态机72h连续运行"""
    states = ['BLOCKED', 'ACTIVE', 'RECOVERY', 'COMPLETE']
    checkpoints = 36  # 每2h一次
    
    state_history = []  # [(timestamp, state), ...]
    anomalies = []
    
    for cp in range(checkpoints):
        ts = f"T+{cp * 2}h"
        current_state = get_current_state()
        
        if current_state not in states:
            anomalies.append({
                'timestamp': ts,
                'state': current_state,
                'reason': '非法状态'
            })
        
        state_history.append((ts, current_state))
    
    # 检查状态流转合法性
    valid_transitions = {
        'BLOCKED': ['ACTIVE'],
        'ACTIVE': ['BLOCKED', 'RECOVERY', 'COMPLETE'],
        'RECOVERY': ['ACTIVE'],
        'COMPLETE': ['BLOCKED', 'ACTIVE']
    }
    
    transition_count = 0
    for i in range(1, len(state_history)):
        prev_state = state_history[i-1][1]
        curr_state = state_history[i][1]
        if prev_state != curr_state:
            transition_count += 1
            if curr_state not in valid_transitions.get(prev_state, []):
                anomalies.append({
                    'timestamp': state_history[i][0],
                    'from': prev_state,
                    'to': curr_state,
                    'reason': '非法流转'
                })
    
    return {
        'total_checkpoints': checkpoints,
        'total_transitions': transition_count,
        'total_anomalies': len(anomalies),
        'states': states,
        'status': 'PASS' if len(anomalies) == 0 else 'FAIL'
    }

result = state_machine_continuity_monitor()

import math

def wilson_ci(successes, total, confidence=0.95):
    """计算Wilson置信区间"""
    z = {0.90: 1.645, 0.95: 1.960, 0.99: 2.576}[confidence]
    
    p_hat = successes / total
    denom = 1 + z**2 / total
    centre = (p_hat + z**2 / (2 * total)) / denom
    margin = (z * math.sqrt(p_hat * (1 - p_hat) / total + z**2 / (4 * total**2))) / denom
    
    lower = centre - margin
    upper = centre + margin
    
    return {
        'n': total,
        'successes': successes,
        'p_hat': p_hat,
        'z': z,
        'lower': lower,
        'upper': upper,
        'width': upper - lower
    }

    """72h持续监控事件丢失率"""

        cp_injected = checkpoint_injected[i]

        cp_lost = cp_injected - cp_consumed
        cp_loss_rate = cp_lost / cp_injected if cp_injected > 0 else 0
        loss_rates.append({
            'checkpoint': i,
            'timestamp': f"T+{i*2}h",
            'injected': cp_injected,
            'consumed': cp_consumed,
            'lost': cp_lost,
            'loss_rate': cp_loss_rate
        })
    
    # 全程丢失率
    total_loss_rate = total_lost / total_injected
    
    # 检查所有检查点

    return {
        'total_injected': total_injected,
        'total_consumed': total_consumed,
        'total_lost': total_lost,
        'total_loss_rate': total_loss_rate,
        'threshold': threshold,
        'checkpoints': checkpoints,
        'all_checkpoints_pass': all_pass,
        'status': 'PASS' if all_pass else 'FAIL'
    }
